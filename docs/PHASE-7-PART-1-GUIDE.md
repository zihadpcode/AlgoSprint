## 🟨 Why this integration

The installed Next.js guide requires `ssr: false` dynamic components to live behind a Client Component boundary. `code-editor.tsx` provides that boundary, leaving Prisma and published content reads on the server. Monaco imports occur in a separate browser component.

Monaco 0.56 provides supported ESM entry points for the editor, features, and language definitions. Its older AMD build is deprecated. This implementation uses the ESM editor with locally bundled features, six syntax definitions, and a same-origin editor worker. See [Monaco's changelog](https://github.com/microsoft/monaco-editor/blob/main/CHANGELOG.md) and [integration concepts](https://github.com/microsoft/monaco-editor/blob/main/README.md).

The editor worker supports editor operations; it does not execute submitted code. Registering a syntax definition adds highlighting, not a compiler, runtime, or full language server. JavaScript, TypeScript, Python, Java, C++ and SQL map to the existing schema enums. Only languages actually supplied in a problem's starter records appear in its selector; the synthetic tests include two languages without changing the five original seeds.

## 🟨 Follow the code

1. The published detail page sends only `problem.starterCode` to `CodeEditor`, keyed by the problem slug. No new database query or hidden-test field is added.
2. `editor-state.ts` maps enum values to readable labels and Monaco IDs. `draftFor` uses nullish coalescing so an intentionally empty string remains empty.
3. `CodeEditor` holds the selected language and a draft map. First selection loads that language's starter; switching back retrieves its current draft. An unrelated rerender, such as saving a note, preserves this state. A different problem key creates new state.
4. The dynamic component displays a loading status while the Monaco bundle loads. An error boundary offers copyable current code if rendering/loading/initialization fails. It never reports the editor as ready when that boundary has failed.
5. `MonacoSurface` creates an editor model and view in an effect. The content listener forwards actual edits through the latest callback. A separate effect applies external values only when they differ from the editor value. A guard prevents programmatic starter loading from looking like user typing.
6. That equality check also avoids calling `setValue` for every typing echo, which would disrupt normal editor undo behavior. Switching language disposes the old view/model and creates a new one with the saved draft.
7. Cleanup disposes the content listener, editor and model. If editor creation fails, the newly created model is also disposed. Strict Mode's setup/cleanup/setup cycle is covered by mocked lifecycle tests.

The editor is 420px high, has responsive layout, wrapped lines, no minimap and explicit screen-reader labeling. `tabFocusMode: true` makes Tab leave the editor by default. The existing starter reference remains read-only so learners can still inspect the original code separately.

## 🟥 Common mistakes and limits

- Using `draft || starter` replaces deliberate empty code; use `draft ?? starter`.
- Recreating Monaco on every keystroke loses selection/undo and leaks resources. Keep lifecycle separate from content synchronization.
- Forgetting model/listener disposal creates hidden state across route changes and Strict Mode mounts.
- Treating highlighting as execution gives misleading results. No solution is evaluated here.
- Claiming a worker works because a build emitted its file overstates validation. Live worker startup and editor behavior remain on the browser checklist.
- Server-only hidden tests and private notes must never become editor props. Only public starter records are passed.

import assert from "node:assert/strict";
import { spawn } from "node:child_process";
import { setTimeout as delay } from "node:timers/promises";
import { randomUUID } from "node:crypto";
import pg from "pg";

const databaseUrl = process.env.TEST_DATABASE_URL;

server.stdout.on("data", collect);

"use client";

import dynamic from "next/dynamic";
import { Component, useId, useState, type ReactNode } from "react";
import { Select } from "@/components/ui/input";
import { CodeBlock } from "@/components/problems/code-block";
import { draftFor, editorLanguages, type EditorDrafts, type EditorStarter } from "./editor-state";

const MonacoSurface = dynamic(() => import("./monaco-surface"), {
  ssr: false,
  loading: () => <p role="status" className="grid h-[420px] place-items-center rounded-xl border border-line text-sm text-muted">Loading code editor…</p>,
});

export function CodeEditor({ starters }: { starters: EditorStarter[] }) {
  const [selected, setSelected] = useState(starters[0]?.language);
  const [drafts, setDrafts] = useState<EditorDrafts>({});
  const id = useId();
  const starter = starters.find((entry) => entry.language === selected);
  if (!starter) return <p className="mt-4 text-sm text-muted">No starter code is available for this problem yet.</p>;
  const language = editorLanguages[starter.language];
  const value = draftFor(starter, drafts);
  return <div className="mt-4 space-y-4">
    <label htmlFor={id} className="block text-sm font-semibold">Editor language</label>
    <Select id={id} value={starter.language} onChange={(event) => {
      const next = starters.find((entry) => entry.language === event.target.value);
      if (next) setSelected(next.language);
    }}>
      {starters.map((entry) => <option key={entry.language} value={entry.language}>{editorLanguages[entry.language].label}</option>)}
    </Select>
    <p className="text-xs leading-6 text-muted">Entry point: <code>{starter.entryPoint}</code>. Drafts stay only on this open page; refreshing or leaving discards them. Code execution is not available yet.</p>
    <EditorBoundary fallback={<div role="alert" className="space-y-4"><p className="text-sm text-warm">The editor could not load. Copy your current code below before reloading.</p><CodeBlock label="Current code" code={value} /></div>}>
      <MonacoSurface language={language.id} value={value} onChange={(code) => {
        setDrafts((previous) => previous[starter.language] === code ? previous : { ...previous, [starter.language]: code });
      }} />
    </EditorBoundary>
    <p className="text-xs leading-6 text-muted">Monaco supports keyboard navigation and screen readers. Use its command palette for accessibility options.</p>
  </div>;
}

class EditorBoundary extends Component<{ children: ReactNode; fallback: ReactNode }, { failed: boolean }> {
  state = { failed: false };
  static getDerivedStateFromError() { return { failed: true }; }
  render() { return this.state.failed ? this.props.fallback : this.props.children; }
}

"use client";

import { useEffect, useRef } from "react";
import * as monaco from "monaco-editor/editor";
import "monaco-editor/features/register.all";
import "monaco-editor/languages/definitions/javascript/register";
import "monaco-editor/languages/definitions/typescript/register";
import "monaco-editor/languages/definitions/python/register";
import "monaco-editor/languages/definitions/java/register";
import "monaco-editor/languages/definitions/cpp/register";
import "monaco-editor/languages/definitions/sql/register";

// Bundled same-origin worker; no CDN loader and no code-execution service.
self.MonacoEnvironment = {
  getWorker() {
    return new Worker(new URL("monaco-editor/editor/editor.worker.js", import.meta.url), { type: "module" });
  },
};

export default function MonacoSurface({ language, value, onChange }: {
  language: string; value: string; onChange: (value: string) => void;
}) {
  const container = useRef<HTMLDivElement>(null);
  const editor = useRef<monaco.editor.IStandaloneCodeEditor | null>(null);
  const callback = useRef(onChange);
  const syncing = useRef(false);
  useEffect(() => { callback.current = onChange; }, [onChange]);
  useEffect(() => {
    if (!container.current) return;
    const model = monaco.editor.createModel("", language);
    let instance: monaco.editor.IStandaloneCodeEditor;
    try { instance = monaco.editor.create(container.current, {
      model, theme: "vs-dark", automaticLayout: true, tabFocusMode: true,
      ariaLabel: `${language} code editor`, accessibilitySupport: "auto",
      minimap: { enabled: false }, fontSize: 14, lineHeight: 24, tabSize: 2,
      scrollBeyondLastLine: false, wordWrap: "on", padding: { top: 16, bottom: 16 },
      fixedOverflowWidgets: true,
    }); } catch (error) { model.dispose(); throw error; }
    editor.current = instance;
    const subscription = instance.onDidChangeModelContent(() => {
      if (!syncing.current) callback.current(instance.getValue());
    });
    return () => {
      subscription.dispose();
      instance.dispose();
      model.dispose();
      editor.current = null;
    };
  }, [language]);
  useEffect(() => {
    const instance = editor.current;
    if (instance && instance.getValue() !== value) {
      syncing.current = true;
      try { instance.setValue(value); } finally { syncing.current = false; }
    }
  }, [value, language]);
  return <div ref={container} className="h-[420px] min-w-0 overflow-hidden rounded-xl border border-line" />;
}
